<?php

$db = new DB();

$db->table('users')->select('id', 'firstname', 'lastname')->where('age', '>', 18)->where('role', '=', 'admin')->order('id', 'desc')->limit(1, 10)->sql();

// select `id`, `firstname`, `lastname` from `users` where `age` > 18 and `role` = 'admin' order by `id` desc limit 0, 10

$db->table(`users`)->select('id', 'firstname')->whereIn('role', ['admin', 'moderator'])->order('id', 'asc')->limit(1, 10)->sql();

// select `id`, `firstname` from `users` where `role` in ('admin', 'moderator') order by `id` asc limit 10, 10;

$db->table('users')->where('id', '=', 1)->update(
	firstname: 'John',
	lastname: 'Doe',
	age: 40
)->sql();

// update `users` set `firstname` = 'John', `lastname` = 'Doe', `age` = 40 where `id` = 1;

$db->table('users')->insert(
	firstname: 'Jane',
	lastname: 'Doe',
	age: 25
)->sql();

// insert into `users` (`firstname`, `lastname`, `age`) values ('Jane', 'Doe', 25);

$db->table('users')->where('id', '=', 1)->delete()->sql();

// delete from `users` where `id` = 1;
